5   package com.jcabi.xml;
6   
7   import java.io.File;
8   import java.io.IOException;
9   import javax.xml.parsers.DocumentBuilder;
10  import org.w3c.dom.Document;
11  import org.xml.sax.SAXException;
12  
13  /**
14   * File source of XML from a file.
15   *
16   * @since 0.32
17   */
18  class FileSource implements DocSource {
19  
20      /**
21       * The file.
22       */
23      private final File file;
24  
25      /**
26       * Public ctor.
27       *
28       * @param file The file
29       */
30      FileSource(final File file) {
31          this.file = file;
32      }
33  
34      @Override
35      public Document apply(final DocumentBuilder builder) throws IOException, SAXException {
36          return builder.parse(this.file);
37      }
38  
39      @Override
40      public long length() {
41          return this.file.length();
42      }
43  }
