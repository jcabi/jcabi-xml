5   package com.jcabi.xml;
6   
7   import java.io.ByteArrayInputStream;
8   import java.io.IOException;
9   import java.nio.charset.StandardCharsets;
10  import javax.xml.parsers.DocumentBuilder;
11  import org.w3c.dom.Document;
12  import org.xml.sax.SAXException;
13  
14  /**
15   * Bytes source of XML.
16   *
17   * @since 0.32
18   */
19  class BytesSource implements DocSource {
20  
21      /**
22       * Bytes of the XML.
23       */
24      private final byte[] xml;
25  
26      /**
27       * Public ctor.
28       *
29       * @param xml Bytes of the XML
30       */
31      BytesSource(final String xml) {
32          this(xml.getBytes(StandardCharsets.UTF_8));
33      }
34  
35      /**
36       * Public ctor.
37       *
38       * @param xml Bytes of the XML
39       */
40      @SuppressWarnings("PMD.ArrayIsStoredDirectly")
41      BytesSource(final byte[] xml) {
42          this.xml = xml;
43      }
44  
45      @Override
46      public Document apply(final DocumentBuilder builder) throws IOException, SAXException {
47          return builder.parse(new ByteArrayInputStream(this.xml));
48      }
49  
50      @Override
51      public long length() {
52          return this.xml.length;
53      }
54  }
